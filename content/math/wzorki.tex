\section{Fibonacci}
$f_0 = 0$, $f_1 = 1$, $f_n = f_{n-1} + f_{n-2}$, dla $n \geq 2$.
\vspace{0.4cm}
\[ f_n = \frac{\alpha^n - \beta ^n}{\sqrt{5}} = \frac{\alpha ^ n - \beta ^ n}{\alpha - \beta}, \]
dla $\alpha = \frac{1 + \sqrt{5}}{2}$, $\beta = \frac{1 - \sqrt{5}}{2}$.
Jako, że $|\beta| < 1$ to 
\[ f_n \approx \frac{1}{\sqrt{5}} \alpha ^ n.\]
Niech $F(x) = \sum_{n=0}^{\infty} f_n x^n$, wtedy 
\[ F(x) = \frac{x}{1 - x - x^2}. \]
\vspace{0.5cm}
\[f_1 + f_2 + \dots + f_n = f_{n+2} - 1\]
\[\gcd(f_n, f_m) = f_{\gcd(n, m)}. \]

\section{Divide and Conquer DP}
Jeżeli
\[ dp(i, j) = \min_{0 \leq k \leq j} dp(i-1, k-1) + C(k, j),\]
oraz 
\[C(a, c) + C(b, d) \leq C(a, d) + C(b, c),\]
dla $a \leq b \leq c \leq d$, to $opt(i, j) \leq opt(i, j+1)$, 
gdzie $opt(i, j)$ to indeks $k$ minimalizujący powyższy wzór.
Można w takim razie użyć divide and conquer do przyspieszenia.

\section{Funkcje arytmetyczne}
\[\varphi(ab) = \varphi(a) \cdot \varphi(b) \cdot \frac{d}{\varphi(d)},\]
gdzie $d = \gcd(a, b)$.
Jeżeli $n = p_1^{\alpha_1} p_2^{\alpha_2} \dots p_k^{\alpha_k}$, to
\[\varphi(n) = n \cdot \left( 1 - \frac{1}{p_1}\right) \cdot \left( 1 - \frac{1}{p_2}\right) \dots \left( 1 - \frac{1}{p_k}\right).\]

\[ \varphi(n) = \sum_{d \mid n} \varphi(d)\]

Niech $\phi(n) = \sum_{i=1}^n \varphi(i)$, wtedy
\[ \phi(n) = \frac{n(n+1)}{2} - \sum_{i=2}^n \phi \left( \left\lfloor \frac{n}{i}\right \rfloor \right)\]
